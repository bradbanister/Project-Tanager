Solution Part 1: Relational Flattening Architecture

Three-Table Design (Unchanged)

-- Core validation and reference tables
CREATE TABLE Documents (
    Id BIGINT PRIMARY KEY,
    DocumentPartitionKey TINYINT,
    DocumentUuid UUID UNIQUE,
    ProjectName VARCHAR(100),
    ResourceName VARCHAR(100),
    ResourceVersion VARCHAR(20)
);

CREATE TABLE References (
    Id BIGINT PRIMARY KEY,
    ParentDocumentPartitionKey TINYINT,
    ParentDocumentId BIGINT,
    ReferencedDocumentId BIGINT,
    ReferentialId UUID,
    FOREIGN KEY (ParentDocumentId) 
        REFERENCES Documents(Id),
    FOREIGN KEY (ReferentialId) 
        REFERENCES Aliases(ReferentialId)
);

CREATE TABLE Aliases (
    Id BIGINT PRIMARY KEY,
    ReferentialId UUID UNIQUE,
    ReferentialPartitionKey TINYINT,
    DocumentId BIGINT,
    DocumentPartitionKey TINYINT,
    FOREIGN KEY (DocumentId) 
        REFERENCES Documents(Id)
);

Flattened Student Tables (New)

-- Relational access tables
CREATE TABLE Student (
    Id BIGINT PRIMARY KEY IDENTITY,
    DocumentUuid UUID NOT NULL,
    DocumentPartitionKey TINYINT NOT NULL,
    StudentUniqueId VARCHAR(100) NOT NULL,
    FirstName VARCHAR(100),
    LastSurname VARCHAR(100),
    BirthDate DATE,
    CONSTRAINT UQ_StudentUniqueId 
        UNIQUE (StudentUniqueId),
    CONSTRAINT FK_Student_Documents 
        FOREIGN KEY (DocumentUuid, DocumentPartitionKey)
        REFERENCES Documents(DocumentUuid, DocumentPartitionKey)
);

CREATE TABLE StudentAddress (
    Id BIGINT PRIMARY KEY IDENTITY,
    Student_Id BIGINT NOT NULL,
    DocumentUuid UUID NOT NULL,
    DocumentPartitionKey TINYINT NOT NULL,
    AddressTypeDescriptor VARCHAR(100),
    City VARCHAR(100),
    StateAbbreviation CHAR(2),
    CONSTRAINT UQ_AddressType 
        UNIQUE (AddressTypeDescriptor),
    CONSTRAINT FK_StudentAddress_Student 
        FOREIGN KEY (Student_Id) REFERENCES Student(Id)
);

CREATE TABLE StudentRace (
    Id BIGINT PRIMARY KEY IDENTITY,
    Student_Id BIGINT NOT NULL,
    DocumentUuid UUID NOT NULL,
    DocumentPartitionKey TINYINT NOT NULL,
    RaceDescriptor VARCHAR(100),
    CONSTRAINT FK_StudentRace_Student 
        FOREIGN KEY (Student_Id) REFERENCES Student(Id)
);

Key Design Decisions (Flattening)

  • ✅ Surrogate keys for simplicity (BIGINT IDENTITY)
  • ✅ Intra-resource FKs only (Flattened tables relate to each other)
  • ✅ No cross-resource FKs (References table handles that)
  • ✅ Each table links to Documents for traceability

Clean Table Design

Example: Student resource flattening

Root Entity Table

CREATE TABLE Student (
    Id BIGINT PRIMARY KEY IDENTITY(1,1),          -- Simple surrogate key
    DocumentUuid UUID NOT NULL,                   -- Link to Documents
    DocumentPartitionKey TINYINT NOT NULL,        -- Partition key
    StudentUniqueId VARCHAR(100) NOT NULL,
    FirstName VARCHAR(100) NOT NULL,
    LastSurname VARCHAR(100) NOT NULL,
    BirthDate DATE,
    -- Flattened nested objects
    Demographics_HispanicLatinoEthnicity BIT,
    Demographics_Sex VARCHAR(20),
    ContactInfo_Email VARCHAR(255),
    CONSTRAINT FK_Student_Documents 
        FOREIGN KEY (DocumentUuid, DocumentPartitionKey) 
        REFERENCES Documents(DocumentUuid, DocumentPartitionKey),
    CONSTRAINT UQ_Student_StudentUniqueId UNIQUE (StudentUniqueId)
);

Child Array Table

CREATE TABLE StudentAddress (
    Id BIGINT PRIMARY KEY IDENTITY(1,1),
    Student_Id BIGINT NOT NULL,      -- Example FK column name, actual will avoid possible name collisions
    DocumentUuid UUID NOT NULL,      -- Document traceability
    DocumentPartitionKey TINYINT NOT NULL,
    AddressTypeDescriptor VARCHAR(100),
    StreetNumberName VARCHAR(255),
    City VARCHAR(100),
    StateAbbreviation CHAR(2),
    CONSTRAINT FK_StudentAddress_Student 
        FOREIGN KEY (Student_Id) REFERENCES Student(Id),
    CONSTRAINT UQ_StudentAddress_Type 
        UNIQUE (AddressTypeDescriptor)
);

Notice: No complex natural keys, no cross-resource FKs, just simple surrogate keys

Database-Side (PostgreSQL)

-- Use native JSON functions
CREATE FUNCTION import_json_document(
    p_json_payload JSONB
) RETURNS INTEGER AS $$
BEGIN
    -- Direct JSON to table mapping
    INSERT INTO Student 
    SELECT * FROM jsonb_populate_record(
        null::Student,
        p_json_payload->>'student'
    );
    
    -- Handle arrays efficiently
    INSERT INTO StudentAddress
    SELECT * FROM jsonb_populate_recordset(
        null::StudentAddress,
        p_json_payload->'addresses'
    );
    
    RETURN 1;
END;
$$ LANGUAGE plpgsql;

Metadata-Driven JSON Flattening

Metadata Schema Defines Everything

{
  "documentType": "Student",
  "entities": [{
    "tableName": "Student",
    "jsonPath": "$.student",
    "properties": [
      {
        "columnName": "StudentUniqueId",
        "dataType": "VARCHAR(100)",
        "jsonPath": "$.studentUniqueId",
        "isPrimaryKey": true
      },
      {
        "columnName": "Demographics_Sex",
        "dataType": "VARCHAR(20)",
        "jsonPath": "$.demographics.sex"  // Nested object
      }
    ]
  }, {
    "tableName": "StudentAddress",
    "jsonPath": "$.student.addresses",  // Array of objects
    "relationships": [{
      "parentTable": "Student",
      "foreignKeyColumn": "Student_Id"
    }]
  }]
}

PostgreSQL JSON Generation

-- Database reconstructs JSON using metadata
CREATE FUNCTION get_student_json(p_student_id TEXT)
RETURNS JSON AS $$
    SELECT json_build_object(
        'studentUniqueId', s.student_unique_id,
        'firstName', s.first_name,
        'demographics', json_build_object(
            'sex', s.demographics_sex
        ),
        'addresses', (
            SELECT json_agg(json_build_object(
                'city', sa.city,
                'stateAbbreviation', sa.state_abbreviation
            ))
            FROM student_address sa
            WHERE sa.student_id = s.id
        )
    )
    FROM student s WHERE s.id = p_student_id;
$$ LANGUAGE sql;

Lightning-Fast JSON Reconstruction

Leverage modern database JSON capabilities for the read path:

SQL Server FOR JSON

CREATE PROCEDURE GetStudentJson
    @DocumentUuid UUID,
    @DocumentPartitionKey TINYINT
AS
SELECT
    s.StudentUniqueId AS 'studentUniqueId',
    s.FirstName AS 'firstName',
    s.LastSurname AS 'lastSurname',
    
    -- Nested demographics object
    JSON_QUERY((
        SELECT
            s.Demographics_HispanicLatinoEthnicity AS 'hispanicLatinoEthnicity',
            s.Demographics_Sex AS 'sex'
        FOR JSON PATH, WITHOUT_ARRAY_WRAPPER
    )) AS 'demographics',
    
    -- Addresses array
    (SELECT
        AddressTypeDescriptor AS 'addressTypeDescriptor',
        City AS 'city',
        StateAbbreviation AS 'stateAbbreviation'
     FROM StudentAddress sa
     WHERE sa.Student_Id = s.Id
     FOR JSON PATH) AS 'addresses'
     
FROM Student s
WHERE s.DocumentUuid = @DocumentUuid 
  AND s.DocumentPartitionKey = @DocumentPartitionKey
FOR JSON PATH, WITHOUT_ARRAY_WRAPPER

The database does the heavy lifting - no application-side JSON assembly needed!

Solution Part 2: Purpose-Built Tooling

The Requirements We're Addressing

  • ⚡ Quick modification of a single field without full document updates
  • 🚀 High-speed bulk data loading that bypasses API overhead
  • ⏱️ Performance-critical operations during tight maintenance windows

Examples of Tooling

  • Field-level PATCH API: Update specific fields without sending entire documents (command line version as well)
  • Batch field updates: Modify the same field across multiple records in one operation
  • SQL-like update syntax: UPDATE Student SET LastName = 'Smith' WHERE StudentUniqueId = '12345'
  • Validation bypass mode: For trusted administrative corrections
